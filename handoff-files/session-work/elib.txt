265:    let tauri::ipc::InvokeBody::Raw(bytes) = request.body() else {

295:#[tauri::command]
296:fn write_chunk(request: tauri::ipc::Request<'_>) -> Result<(), String> {
297:    let tauri::ipc::InvokeBody::Raw(bytes) = request.body() else {
298:        return Err("Nothing to write.".into());
299:    };
300:    let header = |k: &str| {
301:        request
302:            .headers()
303:            .get(k)
304:            .and_then(|v| v.to_str().ok())
305:            .map(str::to_owned)
306:    };
307:    let path = header("x-path").ok_or("No file given.")?;
308:    let path = PathBuf::from(decode(&path));
309:    if !is_work_file(&path) {
310:        return Err("Only the film's work folder can be written.".into());
311:    }
312:    let position: u64 = header("x-position")
313:        .and_then(|v| v.parse().ok())
314:        .unwrap_or(0);
315:    export::write_at(&path, position, bytes)
316:}

319:fn decode(s: &str) -> String {
320:    let bytes = s.as_bytes();
321:    let mut out = Vec::with_capacity(bytes.len());
322:    let mut i = 0;
323:    while i < bytes.len() {
324:        if bytes[i] == b'%' && i + 2 < bytes.len() {
325:            if let Ok(v) = u8::from_str_radix(&s[i + 1..i + 3], 16) {
326:                out.push(v);
327:                i += 3;
328:                continue;
329:            }
330:        }
331:        out.push(bytes[i]);
332:        i += 1;
333:    }
334:    String::from_utf8_lossy(&out).into_owned()
335:}

384:#[tauri::command]
385:async fn find_by_name(folder: String, name: String) -> Option<String> {
386:    tauri::async_runtime::spawn_blocking(move || {
387:        let mut stack = vec![(PathBuf::from(folder), 0)];
388:        while let Some((dir, depth)) = stack.pop() {
389:            let Ok(entries) = std::fs::read_dir(&dir) else {
390:                continue;
391:            };
392:            for e in entries.flatten() {
393:                let p = e.path();
394:                if p.is_dir() {
395:                    if depth < 4 {
396:                        stack.push((p, depth + 1));
397:                    }
398:                } else if p.file_name().and_then(|n| n.to_str()) == Some(name.as_str()) {
399:                    return Some(p.to_string_lossy().into_owned());
400:                }
401:            }
402:        }
403:        None
404:    })
405:    .await
406:    .ok()
407:    .flatten()
408:}
